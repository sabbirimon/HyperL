cmake_minimum_required(VERSION 3.20)
project(HyperL VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
add_library(hyperl_cpu STATIC native/cpu.c)
target_include_directories(hyperl_cpu PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/native/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(hyperl_cpu PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(hyperl_cpu PRIVATE /fp:strict)
else()
  target_compile_options(hyperl_cpu PRIVATE -ffp-contract=off -Wall -Wextra)
endif()
enable_testing()
add_executable(hyperl_cpu_test native/cpu_test.c)
target_link_libraries(hyperl_cpu_test PRIVATE hyperl_cpu)
if(MSVC)
  target_compile_options(hyperl_cpu_test PRIVATE /UNDEBUG)
else()
  target_compile_options(hyperl_cpu_test PRIVATE -UNDEBUG)
endif()
add_test(NAME hyperl_cpu_contract COMMAND hyperl_cpu_test)
install(TARGETS hyperl_cpu EXPORT HyperLTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES native/include/hyperl.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT HyperLTargets FILE HyperLTargets.cmake NAMESPACE HyperL:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/HyperL)
