cmake_minimum_required(VERSION 3.20)
project(HyperL VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
option(HYPERL_WARNINGS_AS_ERRORS "Reject native compiler warnings" ON)
option(HYPERL_SANITIZERS "Build portable C contract with ASan and UBSan" OFF)
option(HYPERL_BENCHMARK_REFERENCE "Build explicitly invoked preallocated CPU baseline" OFF)
add_library(hyperl_cpu STATIC native/cpu.c)
add_library(hyperl_cpu_runtime SHARED native/cpu.c)
set_target_properties(hyperl_cpu_runtime PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
foreach(hl_target IN ITEMS hyperl_cpu hyperl_cpu_runtime)
target_include_directories(${hl_target} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/native/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(${hl_target} PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(${hl_target} PRIVATE /fp:strict /W4)
  if(HYPERL_WARNINGS_AS_ERRORS)
    target_compile_options(${hl_target} PRIVATE /WX)
  endif()
else()
  target_compile_options(${hl_target} PRIVATE -ffp-contract=off -Wall -Wextra)
  if(HYPERL_WARNINGS_AS_ERRORS)
    target_compile_options(${hl_target} PRIVATE -Werror)
  endif()
endif()
endforeach()
enable_testing()
add_executable(hyperl_cpu_test native/cpu_test.c)
target_link_libraries(hyperl_cpu_test PRIVATE hyperl_cpu)
if(MSVC)
  target_compile_options(hyperl_cpu_test PRIVATE /UNDEBUG)
else()
  target_compile_options(hyperl_cpu_test PRIVATE -UNDEBUG)
endif()
add_test(NAME hyperl_cpu_contract COMMAND hyperl_cpu_test)
add_executable(hyperl_cpu_stress native/cpu_contract_stress.c)
target_link_libraries(hyperl_cpu_stress PRIVATE hyperl_cpu)
if(MSVC)
  target_compile_options(hyperl_cpu_stress PRIVATE /UNDEBUG /W4 /WX)
else()
  target_compile_options(hyperl_cpu_stress PRIVATE -UNDEBUG -ffp-contract=off -Wall -Wextra -Werror)
endif()
add_test(NAME hyperl_cpu_contract_stress COMMAND hyperl_cpu_stress)
add_executable(hyperl_cpu_liveness native/cpu_liveness_test.c)
target_include_directories(hyperl_cpu_liveness PRIVATE native/include)
set_target_properties(hyperl_cpu_liveness PROPERTIES C_STANDARD 99)
if(MSVC)
  target_compile_options(hyperl_cpu_liveness PRIVATE /UNDEBUG /fp:strict /W4 /WX)
else()
  target_compile_options(hyperl_cpu_liveness PRIVATE -UNDEBUG -ffp-contract=off -Wall -Wextra -Werror)
endif()
add_test(NAME hyperl_cpu_liveness COMMAND hyperl_cpu_liveness)
if(HYPERL_BENCHMARK_REFERENCE)
  add_library(hyperl_benchmark_reference SHARED native/benchmark_reference.c)
  set_target_properties(hyperl_benchmark_reference PROPERTIES C_STANDARD 99 WINDOWS_EXPORT_ALL_SYMBOLS ON)
  if(MSVC)
    target_compile_options(hyperl_benchmark_reference PRIVATE /fp:strict /W4 /WX)
  else()
    target_compile_options(hyperl_benchmark_reference PRIVATE -ffp-contract=off -Wall -Wextra -Werror)
  endif()
endif()
if(HYPERL_SANITIZERS)
  if(MSVC OR NOT CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "ASan/UBSan requires Clang or GCC")
  endif()
  foreach(hl_target IN ITEMS hyperl_cpu hyperl_cpu_runtime hyperl_cpu_test hyperl_cpu_stress hyperl_cpu_liveness)
    target_compile_options(${hl_target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${hl_target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
if(APPLE)
  enable_language(OBJCXX)
  add_executable(hyperl_metal native/metal_bridge.mm)
  set_target_properties(hyperl_metal PROPERTIES OUTPUT_NAME hyperl-metal OBJCXX_STANDARD 17)
  target_compile_options(hyperl_metal PRIVATE -fobjc-arc -Wall -Wextra -Wno-deprecated-declarations)
  target_link_libraries(hyperl_metal PRIVATE "-framework Foundation" "-framework Metal")
  add_test(NAME hyperl_metal_probe COMMAND hyperl_metal --probe)
endif()
install(TARGETS hyperl_cpu hyperl_cpu_runtime EXPORT HyperLTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES native/include/hyperl.h native/include/hyperl_contract.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT HyperLTargets FILE HyperLTargets.cmake NAMESPACE HyperL:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/HyperL)

# Separate ABI for verified storage erasures; existing CPU numerical ABI is unchanged.
add_library(hyperl_erasure SHARED native/erasure.c)
target_include_directories(hyperl_erasure PUBLIC native/include)
set_target_properties(hyperl_erasure PROPERTIES C_STANDARD 99 WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_executable(hyperl_erasure_test native/erasure_test.c)
target_link_libraries(hyperl_erasure_test PRIVATE hyperl_erasure)
if(MSVC)
  target_compile_options(hyperl_erasure PRIVATE /W4 /WX)
  target_compile_options(hyperl_erasure_test PRIVATE /UNDEBUG /W4 /WX)
else()
  target_compile_options(hyperl_erasure PRIVATE -Wall -Wextra -Werror)
  target_compile_options(hyperl_erasure_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
endif()
if(HYPERL_SANITIZERS)
  foreach(hl_erasure_target IN ITEMS hyperl_erasure hyperl_erasure_test)
    target_compile_options(${hl_erasure_target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${hl_erasure_target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
add_test(NAME hyperl_erasure_contract COMMAND hyperl_erasure_test)
# HyperL Native Engine Core & SIMD Kernels
add_library(hyperl_engine SHARED native/engine.c native/kernels.c)
add_library(hyperl_engine_static STATIC native/engine.c native/kernels.c)
set_target_properties(hyperl_engine PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON C_STANDARD 11 POSITION_INDEPENDENT_CODE ON)
set_target_properties(hyperl_engine_static PROPERTIES C_STANDARD 11 POSITION_INDEPENDENT_CODE ON)

foreach(hl_eng IN ITEMS hyperl_engine hyperl_engine_static)
  target_include_directories(${hl_eng} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/native> $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/native/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  if(MSVC)
    target_compile_options(${hl_eng} PRIVATE /W4)
  else()
    target_compile_options(${hl_eng} PRIVATE -Wall -Wextra -O3)
  endif()
endforeach()

add_executable(hyperl_engine_test native/engine_test.c)
target_link_libraries(hyperl_engine_test PRIVATE hyperl_engine_static)
target_include_directories(hyperl_engine_test PRIVATE native/include native)
if(NOT MSVC)
  target_compile_options(hyperl_engine_test PRIVATE -Wall -Wextra -O3)
endif()
add_test(NAME hyperl_native_engine_test COMMAND hyperl_engine_test)

install(TARGETS hyperl_erasure ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES native/include/hyperl_erasure.h native/include/hyperl_engine.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})


target_compile_definitions(hyperl_engine PRIVATE HYPERL_BUILDING_DLL)
target_compile_definitions(hyperl_engine_static PUBLIC HYPERL_STATIC)
if(UNIX AND NOT APPLE)
  target_link_libraries(hyperl_engine PRIVATE m)
  target_link_libraries(hyperl_engine_static PUBLIC m)
endif()
install(TARGETS hyperl_engine hyperl_engine_static)
install(FILES native/include/hyperl.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(HYPERL_SANITIZERS)
  foreach(target IN ITEMS hyperl_engine hyperl_engine_static hyperl_engine_test)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
