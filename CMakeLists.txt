cmake_minimum_required(VERSION 3.20)
project(HyperL VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
add_library(hyperl_cpu STATIC native/cpu.c)
add_library(hyperl_cpu_runtime SHARED native/cpu.c)
set_target_properties(hyperl_cpu_runtime PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
foreach(hl_target IN ITEMS hyperl_cpu hyperl_cpu_runtime)
target_include_directories(${hl_target} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/native/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(${hl_target} PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(${hl_target} PRIVATE /fp:strict)
else()
  target_compile_options(${hl_target} PRIVATE -ffp-contract=off -Wall -Wextra)
endif()
endforeach()
enable_testing()
add_executable(hyperl_cpu_test native/cpu_test.c)
target_link_libraries(hyperl_cpu_test PRIVATE hyperl_cpu)
if(MSVC)
  target_compile_options(hyperl_cpu_test PRIVATE /UNDEBUG)
else()
  target_compile_options(hyperl_cpu_test PRIVATE -UNDEBUG)
endif()
add_test(NAME hyperl_cpu_contract COMMAND hyperl_cpu_test)
if(APPLE)
  enable_language(OBJCXX)
  add_executable(hyperl_metal native/metal_bridge.mm)
  set_target_properties(hyperl_metal PROPERTIES OUTPUT_NAME hyperl-metal OBJCXX_STANDARD 17)
  target_compile_options(hyperl_metal PRIVATE -fobjc-arc -Wall -Wextra -Wno-deprecated-declarations)
  target_link_libraries(hyperl_metal PRIVATE "-framework Foundation" "-framework Metal")
  add_test(NAME hyperl_metal_probe COMMAND hyperl_metal --probe)
endif()
install(TARGETS hyperl_cpu hyperl_cpu_runtime EXPORT HyperLTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES native/include/hyperl.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT HyperLTargets FILE HyperLTargets.cmake NAMESPACE HyperL:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/HyperL)
