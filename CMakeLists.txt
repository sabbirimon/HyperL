cmake_minimum_required(VERSION 3.20)
project(HyperL VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
option(HYPERL_WARNINGS_AS_ERRORS "Reject native compiler warnings" ON)
option(HYPERL_SANITIZERS "Build portable C contract with ASan and UBSan" OFF)
option(HYPERL_BENCHMARK_REFERENCE "Build explicitly invoked preallocated CPU baseline" OFF)
add_library(hyperl_cpu STATIC native/cpu.c)
add_library(hyperl_cpu_runtime SHARED native/cpu.c)
set_target_properties(hyperl_cpu_runtime PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
foreach(hl_target IN ITEMS hyperl_cpu hyperl_cpu_runtime)
target_include_directories(${hl_target} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/native/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(${hl_target} PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(${hl_target} PRIVATE /fp:strict /W4)
  if(HYPERL_WARNINGS_AS_ERRORS)
    target_compile_options(${hl_target} PRIVATE /WX)
  endif()
else()
  target_compile_options(${hl_target} PRIVATE -ffp-contract=off -Wall -Wextra)
  if(HYPERL_WARNINGS_AS_ERRORS)
    target_compile_options(${hl_target} PRIVATE -Werror)
  endif()
endif()
endforeach()
enable_testing()
add_executable(hyperl_cpu_test native/cpu_test.c)
target_link_libraries(hyperl_cpu_test PRIVATE hyperl_cpu)
if(MSVC)
  target_compile_options(hyperl_cpu_test PRIVATE /UNDEBUG)
else()
  target_compile_options(hyperl_cpu_test PRIVATE -UNDEBUG)
endif()
add_test(NAME hyperl_cpu_contract COMMAND hyperl_cpu_test)
add_executable(hyperl_cpu_stress native/cpu_contract_stress.c)
target_link_libraries(hyperl_cpu_stress PRIVATE hyperl_cpu)
if(MSVC)
  target_compile_options(hyperl_cpu_stress PRIVATE /UNDEBUG /W4 /WX)
else()
  target_compile_options(hyperl_cpu_stress PRIVATE -UNDEBUG -ffp-contract=off -Wall -Wextra -Werror)
endif()
add_test(NAME hyperl_cpu_contract_stress COMMAND hyperl_cpu_stress)
if(HYPERL_BENCHMARK_REFERENCE)
  add_library(hyperl_benchmark_reference SHARED native/benchmark_reference.c)
  set_target_properties(hyperl_benchmark_reference PROPERTIES C_STANDARD 99 WINDOWS_EXPORT_ALL_SYMBOLS ON)
  if(MSVC)
    target_compile_options(hyperl_benchmark_reference PRIVATE /fp:strict /W4 /WX)
  else()
    target_compile_options(hyperl_benchmark_reference PRIVATE -ffp-contract=off -Wall -Wextra -Werror)
  endif()
endif()
if(HYPERL_SANITIZERS)
  if(MSVC OR NOT CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "ASan/UBSan requires Clang or GCC")
  endif()
  foreach(hl_target IN ITEMS hyperl_cpu hyperl_cpu_runtime hyperl_cpu_test hyperl_cpu_stress)
    target_compile_options(${hl_target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${hl_target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
if(APPLE)
  enable_language(OBJCXX)
  add_executable(hyperl_metal native/metal_bridge.mm)
  set_target_properties(hyperl_metal PROPERTIES OUTPUT_NAME hyperl-metal OBJCXX_STANDARD 17)
  target_compile_options(hyperl_metal PRIVATE -fobjc-arc -Wall -Wextra -Wno-deprecated-declarations)
  target_link_libraries(hyperl_metal PRIVATE "-framework Foundation" "-framework Metal")
  add_test(NAME hyperl_metal_probe COMMAND hyperl_metal --probe)
endif()
install(TARGETS hyperl_cpu hyperl_cpu_runtime EXPORT HyperLTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES native/include/hyperl.h native/include/hyperl_contract.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT HyperLTargets FILE HyperLTargets.cmake NAMESPACE HyperL:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/HyperL)
